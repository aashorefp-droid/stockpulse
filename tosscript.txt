# ─── ClosePos: normalize close within a bar's range ──────────
#     Returns a value 0-1 (0 = at low, 1 = at high).
#     Used for buyer/seller conviction on current and prev bar.
script ClosePos {
    input h = high;
    input l = low;
    input c = close;
    def r = h - l;
    plot Result = if r > 0 then (c - l) / r else 0.5;
}

# ─── DateYear / DateMonth / DateDay ──────────────────────────
#     Extract individual components from a YYYYMMDD integer.
#     Used for conviction, signal, phase, and Wyckoff date labels.
script DateYear {
    input rawDate = 0;
    plot Result = Floor(rawDate / 10000);
}

script DateMonth {
    input rawDate = 0;
    plot Result = Floor((rawDate % 10000) / 100);
}

script DateDay {
    input rawDate = 0;
    plot Result = rawDate % 100;
}

# ─── WeinScore: sum Stan Weinstein breakout criteria ─────────
#     Accepts six 0/1 boolean inputs and returns their total (0-6).
script WeinScore {
    input maTurningUp   = 0;
    input ma10AboveMA30 = 0;
    input rsPositive    = 0;
    input rsImproving   = 0;
    input volBuilding   = 0;
    input nearHigh      = 0;
    plot Result = maTurningUp + ma10AboveMA30 + rsPositive
                + rsImproving + volBuilding   + nearHigh;
}

# ─── GapSizePct: FVG gap size as a percentage of price ───────
#     Used for both bull and bear FVG size labels.
script GapSizePct {
    input gapTop   = 0;
    input gapBot   = 0;
    input refPrice = close;
    def   size     = gapTop - gapBot;
    plot  Result   = if refPrice != 0 then (size / refPrice) * 100 else 0;
}

# ─── FVGBullTop / FVGBullBot / FVGBearTop / FVGBearBot ───────
#     Select the correct FVG boundary price based on which of the
#     five candidate patterns (f0-f4) fired on this bar.
script FVGBullTop {
    input f0 = no; input f1 = no; input f2 = no;
    input f3 = no; input f4 = no;
    plot Result =
        if f0 then low[1]  else
        if f1 then low[2]  else
        if f2 then low[3]  else
        if f3 then low[4]  else
        if f4 then low[5]  else Double.NaN;
}

script FVGBullBot {
    input f0 = no; input f1 = no; input f2 = no;
    input f3 = no; input f4 = no;
    plot Result =
        if f0 then high[3] else
        if f1 then high[4] else
        if f2 then high[5] else
        if f3 then high[6] else
        if f4 then high[7] else Double.NaN;
}

script FVGBearTop {
    input f0 = no; input f1 = no; input f2 = no;
    input f3 = no; input f4 = no;
    plot Result =
        if f0 then low[3]  else
        if f1 then low[4]  else
        if f2 then low[5]  else
        if f3 then low[6]  else
        if f4 then low[7]  else Double.NaN;
}

script FVGBearBot {
    input f0 = no; input f1 = no; input f2 = no;
    input f3 = no; input f4 = no;
    plot Result =
        if f0 then high[1] else
        if f1 then high[2] else
        if f2 then high[3] else
        if f3 then high[4] else
        if f4 then high[5] else Double.NaN;
}

# ─── 1. PRICE & SWING CALCULATIONS ──────────────────────────
def HH           = Highest(high, lookbackPeriod);
def LL           = Lowest(low,  lookbackPeriod);
def swingRange   = HH - LL;
def currentPrice = close;

# ─── 2. VOLUME LOGIC ────────────────────────────────────────
def avgVol        = Average(volume, 20);
def highVolume    = volume > (avgVol * fibVolThreshold);
def avgVolFVG     = Average(volume, fvgVolLookback);
def highVolumeFVG = volume > (avgVolFVG * fvgVolThreshold);

# ─── 3. BAR TRACKING / TREND DIRECTION ──────────────────────
def barHH          = if high == HH then BarNumber() else barHH[1];
def barLL          = if low  == LL then BarNumber() else barLL[1];
def isBearishSwing = barLL > barHH;
def isBullishSwing = barHH > barLL;

# ─── 4. FIBONACCI TARGET CALCULATIONS ───────────────────────
def bearT1 = LL;
def bearT2 = LL - (swingRange * 0.272);
def bearT3 = LL - (swingRange * 0.618);

# ─── 5. FIBONACCI STATE DEFINITIONS ─────────────────────────
def isBearExt      = isBearishSwing and currentPrice < LL;
def isBearRetrace  = isBearishSwing and currentPrice >= LL and currentPrice <= HH;
def hitLowRecently = Lowest(low, 3) <= LL;

# ─── 6. BUYER / SELLER CONVICTION ───────────────────────────
def closePosition    = ClosePos(h = high,    l = low,    c = close);
def prevClosePos     = ClosePos(h = high[1], l = low[1], c = close[1]);
def buyerConviction  = highVolume and closePosition >= 0.5;
def sellerConviction = highVolume and closePosition <  0.5;
def strongBuyers     = buyerConviction  and prevClosePos >= 0.5;
def strongSellers    = sellerConviction and prevClosePos <  0.5;

# ─── 6B. DATE TRACKING FOR CONVICTION BARS ───────────────────
def buyerConvBar     = if buyerConviction  then BarNumber() else buyerConvBar[1];
def sellerConvBar    = if sellerConviction then BarNumber() else sellerConvBar[1];

def buyerConvOffset  = BarNumber() - buyerConvBar;
def sellerConvOffset = BarNumber() - sellerConvBar;

def buyerConvRawDate  = GetValue(GetYYYYMMDD(), buyerConvOffset);
def buyerConvYear     = DateYear(rawDate  = buyerConvRawDate);
def buyerConvMonth    = DateMonth(rawDate = buyerConvRawDate);
def buyerConvDay      = DateDay(rawDate   = buyerConvRawDate);

def sellerConvRawDate = GetValue(GetYYYYMMDD(), sellerConvOffset);
def sellerConvYear    = DateYear(rawDate  = sellerConvRawDate);
def sellerConvMonth   = DateMonth(rawDate = sellerConvRawDate);
def sellerConvDay     = DateDay(rawDate   = sellerConvRawDate);

def buyerConvOpen    = GetValue(open, buyerConvOffset);
def sellerConvOpen   = GetValue(open, sellerConvOffset);

# ─── 6C. STRENGTH STATE AT CONVICTION TIME ───────────────────
def buyerConvStrong  = if buyerConviction  then strongBuyers  else buyerConvStrong[1];
def sellerConvStrong = if sellerConviction then strongSellers else sellerConvStrong[1];

# ─── 7. ZIGZAG / MARKET PHASE ───────────────────────────────
def priceH = high;
def priceL = low;
def ATR    = reference ATR(length = ATRlength);
def reversalAmount = if (close * zigzagpercent / 100) > Max(zigzagamount < ATRreversalfactor * ATR, zigzagamount)
                     then (close * zigzagpercent / 100)
                     else if zigzagamount < ATRreversalfactor * ATR
                     then ATRreversalfactor * ATR
                     else zigzagamount;

def zzBarNum   = BarNumber();
def zzBarCount = HighestAll(If(IsNaN(close), 0, zzBarNum));

rec zzState = {default init, undefined, uptrend, downtrend};
rec zzMinMax;

if (GetValue(zzState, 1) == GetValue(zzState.init, 0)) {
    zzMinMax = close;
    zzState  = zzState.undefined;
} else if (GetValue(zzState, 1) == GetValue(zzState.undefined, 0)) {
    if (close <= GetValue(zzMinMax, 1) - reversalAmount) {
        zzState  = zzState.downtrend;
        zzMinMax = priceL;
    } else if (close >= GetValue(zzMinMax, 1) + reversalAmount) {
        zzState  = zzState.uptrend;
        zzMinMax = priceH;
    } else {
        zzState  = zzState.undefined;
        zzMinMax = GetValue(zzMinMax, 1);
    }
} else if (GetValue(zzState, 1) == GetValue(zzState.uptrend, 0)) {
    if (close <= GetValue(zzMinMax, 1) - reversalAmount) {
        zzState  = zzState.downtrend;
        zzMinMax = priceL;
    } else {
        zzState  = zzState.uptrend;
        zzMinMax = Max(priceH, GetValue(zzMinMax, 1));
    }
} else {
    if (close >= GetValue(zzMinMax, 1) + reversalAmount) {
        zzState  = zzState.uptrend;
        zzMinMax = priceH;
    } else {
        zzState  = zzState.downtrend;
        zzMinMax = Min(priceL, GetValue(zzMinMax, 1));
    }
}

def zzCalc    = GetValue(zzState, 0) != GetValue(zzState, 1) and zzBarNum >= 1;
def zzFuture  = zzBarCount - zzBarNum;
def zzTmpLast;

if (zzCalc) {
    if (zzFuture >= 1 and GetValue(zzState, 0) == GetValue(zzState, -1)) {
        zzTmpLast = fold lastPeriodBarI = 2 to zzFuture + 1 with lastPeriodBarAcc = 1
            while lastPeriodBarAcc > 0
            do if (GetValue(zzState, 0) != GetValue(zzState, -lastPeriodBarI))
                then -lastPeriodBarAcc
                else lastPeriodBarAcc + 1;
    } else {
        zzTmpLast = 0;
    }
} else {
    zzTmpLast = Double.NaN;
}

def zzLastBar = if (!IsNaN(zzTmpLast)) then -AbsValue(zzTmpLast) else -zzFuture;

rec zzPriceLevel;
rec zzPoints;

if (zzState == zzState.uptrend and zzCalc) {
    zzPriceLevel = fold barWithMaxI = zzLastBar to 1 with maxAcc = zzMinMax
        do Max(maxAcc, GetValue(zzMinMax, barWithMaxI));
    zzPoints = fold maxPtI = zzLastBar to 1 with maxPtAcc = Double.NaN
        while IsNaN(maxPtAcc)
        do if (GetValue(priceH, maxPtI) == zzPriceLevel)
            then maxPtI else maxPtAcc;
} else if (zzState == zzState.downtrend and zzCalc) {
    zzPriceLevel = fold barWithMinI = zzLastBar to 1 with minAcc = zzMinMax
        do Min(minAcc, GetValue(zzMinMax, barWithMinI));
    zzPoints = fold minPtI = zzLastBar to 1 with minPtAcc = Double.NaN
        while IsNaN(minPtAcc)
        do if (GetValue(priceL, minPtI) == zzPriceLevel)
            then minPtI else minPtAcc;
} else if (!zzCalc and (zzState == zzState.uptrend or zzState == zzState.downtrend)) {
    zzPriceLevel = GetValue(zzPriceLevel, 1);
    zzPoints     = GetValue(zzPoints, 1) + 1;
} else {
    zzPoints     = 1;
    zzPriceLevel = GetValue(close, zzPoints);
}

rec zzSave = if !IsNaN(if (zzBarNum == zzBarCount or zzBarNum == 1)
                       then if zzState == zzState.uptrend then priceH else priceL
                       else if (zzPoints == 0) then zzPriceLevel else Double.NaN)
             then if (zzBarNum == zzBarCount or zzBarNum == 1)
                  then if IsNaN(zzBarNum[-1]) and zzState == zzState.uptrend then priceH else priceL
                  else zzPriceLevel
             else GetValue(zzSave, 1);

def zzChg  = (if zzBarNum == zzBarCount and zzPoints < 0 then priceH
              else if zzBarNum == zzBarCount and zzPoints > 0 then priceL
              else zzPriceLevel) - GetValue(zzSave, 1);
def isUp   = zzChg >= 0;
rec isConf = AbsValue(zzChg) >= reversalAmount or
             (IsNaN(GetValue(if (zzBarNum == zzBarCount or zzBarNum == 1)
                             then if zzState == zzState.uptrend then priceH else priceL
                             else if (zzPoints == 0) then zzPriceLevel else Double.NaN, 1))
              and GetValue(isConf, 1));

# ─── 9. AGIG SIGNAL GRADER + CURRENT BIAS ───────────────────

def EIH = ZigZagHighLow("price h" = biasPriceH, "price l" = biasPriceL, "percentage reversal" = .01, "absolute reversal" = .05, "atr length" = 5, "atr reversal" = atrReversal).lastH;

def lastLowBar      = if !IsNaN(EIL) then BarNumber() else lastLowBar[1];
def lastHighBar     = if !IsNaN(EIH) then BarNumber() else lastHighBar[1];
def isLastSignalLow = lastLowBar > lastHighBar;
def signalBar       = if isLastSignalLow then lastLowBar else lastHighBar;
def sigOffset       = BarNumber() - signalBar;

def sigClose = GetValue(close,  sigOffset);
def sigOpen  = GetValue(open,   sigOffset);
def sigVol   = GetValue(volume, sigOffset);
def prevVol  = GetValue(volume, sigOffset + 1);

def sigRawDate = GetValue(GetYYYYMMDD(), sigOffset);
def sigYear    = DateYear(rawDate  = sigRawDate);
def sigMonth   = DateMonth(rawDate = sigRawDate);
def sigDay     = DateDay(rawDate   = sigRawDate);

def isStrongPrice = if isLastSignalLow then sigClose > sigOpen else sigClose < sigOpen;
def isHighVol     = sigVol > prevVol;
def qualityScore  = 1 + (if isStrongPrice then 1 else 0) + (if isHighVol then 1 else 0);

def biasStrength  = Round(AbsValue(close - sigClose) / sigClose * 100, 2);
def isBullishBias = isLastSignalLow  and close >= sigClose;
def isBearishBias = !isLastSignalLow and close <= sigClose;
def isNeutralBias = !isBullishBias and !isBearishBias;
def biasLabel     = if isBullishBias then 1 else if isBearishBias then 2 else 3;

# ─── 10. STAN WEINSTEIN CALCULATIONS ────────────────────────
def MA30 = Average(close, ma30Period);
def MA10 = Average(close, ma10Period);

def ma30Slope = (MA30 - MA30[10]) / MA30[10];
def maIsFlat  = AbsValue(ma30Slope) < flatnessPct;

def high52        = Highest(high, highLookback);
def low52         = Lowest(low,  lowLookback);
def range52       = high52 - low52;
def pricePosition = if range52 > 0 then (close - low52) / range52 * 100 else 0;
def distFromHigh  = (high52 - close) / close * 100;

def stockChange = close / close[rsPeriod];
def indexChange = close(compSymbol) / close(compSymbol)[rsPeriod];
def RS          = stockChange / indexChange - 1;
def rsImproving = RS > RS[4] and RS > RS[8];

def avgVol10       = Average(volume, volumePeriod);
def avgVol4        = Average(volume, 4);
def volumeBuilding = avgVol4 > avgVol10 * 1.1;

def ma10AboveMA30  = MA10 > MA30;
def maTurningUp    = MA30 > MA30[2] and MA30 > MA30[4];

def nearMA30 = close > MA30 * 0.90 and close < MA30 * 1.15;
def inStage1 = nearMA30 and maIsFlat;

def score_ma30Curling = if maTurningUp    then 1 else 0;
def score_ma10Cross   = if ma10AboveMA30  then 1 else 0;
def score_rsPositive  = if RS > 0         then 1 else 0;
def score_rsImproving = if rsImproving    then 1 else 0;
def score_volBuilding = if volumeBuilding then 1 else 0;
def score_nearHigh    = if distFromHigh < 10 then 1 else 0;

def breakoutScore = WeinScore(
    maTurningUp   = score_ma30Curling,
    ma10AboveMA30 = score_ma10Cross,
    rsPositive    = score_rsPositive,
    rsImproving   = score_rsImproving,
    volBuilding   = score_volBuilding,
    nearHigh      = score_nearHigh);

def bearishFVG = bearFVG0 or bearFVG1 or bearFVG2 or bearFVG3 or bearFVG4;

def bullGapSize = bullFVGTop - bullFVGBottom;
def bearGapSize = bearFVGTop - bearFVGBottom;
def bullSizePct = GapSizePct(gapTop = bullFVGTop, gapBot = bullFVGBottom, refPrice = close);
def bearSizePct = GapSizePct(gapTop = bearFVGTop, gapBot = bearFVGBottom, refPrice = close);

def validBullFVG = bullishFVG and !IsNaN(bullFVGTop) and !IsNaN(bullFVGBottom) and bullSizePct >= 0.01;
def validBearFVG = bearishFVG and !IsNaN(bearFVGTop) and !IsNaN(bearFVGBottom) and bearSizePct >= 0.01;

def lastBullTop    = if validBullFVG then bullFVGTop    else lastBullTop[1];

def lastBearTop    = if validBearFVG then bearFVGTop    else lastBearTop[1];
def lastBearBottom = if validBearFVG then bearFVGBottom else lastBearBottom[1];

# ─── 12. FVG ENTRY / EXIT SIGNALS ───────────────────────────

def bullFVGFilled = !IsNaN(lastBullBottom) and close < lastBullBottom;
def bearFVGFilled = !IsNaN(lastBearTop)    and close > lastBearTop;

# ─── VOLUME BIAS (Schwab Volume Signal Analyzer) ────────────
def vb_avgVol         = Average(volume, volAvgLength);
def vb_volRatio       = if vb_avgVol > 0 then volume / vb_avgVol else 0;
def vb_highVol        = vb_volRatio >= volThreshold;
def vb_priceMA        = Average(close, volPriceAvgLen);
def vb_uptrend        = close > vb_priceMA and close > close[1];
def vb_downtrend      = close < vb_priceMA and close < close[1];
def vb_recentAvgVol   = Average(volume, 10);
def vb_olderAvgVol    = Average(volume[10], 10);
def vb_volIncreasing  = vb_recentAvgVol > vb_olderAvgVol;
def vb_volDecreasing  = vb_recentAvgVol < vb_olderAvgVol;
def vb_highestHigh    = Highest(high[1], 20);
def vb_lowestLow      = Lowest(low[1],  20);
def vb_upsideBreak    = close > vb_highestHigh;
def vb_downsideBreak  = close < vb_lowestLow;
def vb_bullBreakout   = vb_upsideBreak   and vb_highVol;
def vb_bullTrend      = vb_uptrend       and vb_volIncreasing;
def vb_weakBull       = vb_uptrend       and vb_volDecreasing;
def vb_bearBreakout   = vb_downsideBreak and vb_highVol;
def vb_bearTrend      = vb_downtrend     and vb_volIncreasing;
def vb_weakBear       = vb_downtrend     and vb_volDecreasing;
def vb_score =
    if vb_bullBreakout then 3
    else if vb_bullTrend then 2
    else if vb_weakBull then 1
    else if vb_bearBreakout then -3
    else if vb_bearTrend then -2
    else if vb_weakBear then -1
    else 0;

# ── PRICE CONTRADICTION DETECTION ────────────────────────────
# For LONG: Stop must be BELOW entry, Target must be ABOVE entry
# For SHORT: Stop must be ABOVE entry, Target must be BELOW entry

# ENTER LONG contradiction check
def enterLongStop = lastBullBottom - (lastBullTop - lastBullBottom);
def enterLongTarget = lastBullTop;
def enterLongContro = bullishEntry and (enterLongStop >= close or enterLongTarget <= close);

# ENTER SHORT contradiction check
def enterShortStop = lastBearTop + (lastBearTop - lastBearBottom);
def enterShortTarget = lastBearBottom;
def enterShortContro = bearishEntry and (enterShortStop <= close or enterShortTarget >= close);

# HOLD LONG contradiction check (Stop = ema50, Target = HH)
def holdLongContro = (isBullishBias or is_uptrend) and (ema50 >= close or HH <= close);

# HOLD SHORT contradiction check (Stop = ema50, Target = LL)
def holdShortContro = (isBearishBias or is_downtrend) and (ema50 <= close or LL >= close);

# Wyckoff LONG contradiction check
def wyLongContro = wyBuyActive and (wy_springLow >= close or wy_upthrustHigh <= close);

# Wyckoff SHORT contradiction check
def wySellContro = wySellActive and (wy_upthrustHigh <= close or wy_supportPrice >= close);

# Any contradiction active

def ix_djx  = close("$DJI");
def ix_ndx  = close("COMP");

def ix_ndxChg = if ix_ndx[5] > 0 then (ix_ndx - ix_ndx[5]) / ix_ndx[5] * 100 else 0;
